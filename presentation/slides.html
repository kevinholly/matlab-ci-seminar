APPENDIX C | BUILDFILE.M
MATLAB owns the task dependencies
function plan = buildfile import matlab.buildtool.tasks.* plan = buildplan; plan("check") = CodeIssuesTask("src", ... Configuration="factory",ErrorThreshold=0); testTask = TestTask(Dependencies="check", ... SourceFiles="src", ... TestResults=["artifacts/tests/results.xml", ... "artifacts/tests/report.html"]); testTask = addCodeCoverage(testTask, ... "artifacts/coverage/report.html",Metrics="statement"); plan("test") = testTask; plan("package") = PackageTask(Dependencies="test"); plan.DefaultTasks = "package"; end
